def withdraw(balance, amount)
  fee = 110
  if balance >= amount + fee
   puts "#{amount}円引き落としました。残高は#{ balance - amount - fee}"
  else
   puts "残高不足です"
  end
end

balance = 100000
puts "いくら引き落としますか？（手数料110円かかります）"
money = gets.to_i  
withdraw(balance, money)



【PHP版】

function withdraw($balance, $amount) {
    $fee = 110;
    if ($balance >= $amount + $fee) {
        $remaining = $balance - $amount - $fee;
        echo "{$amount}円引き落としました。残高は{$remaining}\n";
    } else {
        echo "残高不足です。\n";
    }
}
