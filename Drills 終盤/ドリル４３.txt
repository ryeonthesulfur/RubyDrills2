def get_week(year, month, day)
    weeks = ["日", "月", "火", "水", "木", "金", "土"]
    days = 0

    year_index = year - 1
    while year_index > 0 do
      if get_days(year_index, 2) == 29
        days += 366
      else
        days += 365
      end
      year_index -= 1
    end

    month_index = month - 1
    while month_index > 0 do
    month_days = get_days(year, month_index)
    days += month_days
    month_index -= 1
    end

    days += day
    return weeks[days % 7]
end
