■ クラス変数で配列を管理する
  @@foods = []                   → クラス変数として配列を持つ
  @@foods << food                → どのメソッドからでも引数なしで追加できる
  今まで: register_data(cart) → 引数で配列を持ち回る
  今回: @@foods に直接追加 → 引数不要

■ クラスメソッドでデータ操作
  def self.input ... end         → Food.input で呼ぶクラスメソッド
  def self.show_all_calory       → Food.show_all_calory で呼ぶ
  インスタンスを作る処理もクラスメソッド内に書ける

■ インスタンスを配列に入れるパターン
  food = Food.new(name, calory)
  @@foods << food                → ハッシュではなくインスタンスを配列に追加
  @@foods.each do |food|
    food.name / food.calory      → attr_reader でゲッターを使って取り出す

■ カロリー合計の計算
  all_calory = 0
  all_calory += food.calory      → each の中でインスタンスの値を足し込む

■ attr_reader の役割
  attr_reader :name, :calory     → @name / @calory を外から読めるようにする
  @@foods.each の中で food.name と書けるのは attr_reader があるから

■ PHP 対応メモ
  @@foods = []                   → PHP では private static $foods = []
  @@foods << food                → PHP では self::$foods[] = $food
  def self.input                 → PHP では public static function input()
  Food.input                     → PHP では Food::input()

  class Food
    @@foods = []
   attr_reader :name, :calory

   def self.input
    puts "料理名を入力してください"
    name = gets.chomp
    puts "カロリーを入力してください"
    calory = gets.to_i

    food = Food.new(name, calory)
    @@foods << food
   end

   def self.show_all_calory
    all_calory = 0
    line =  "-----------------------------"
    puts line

    @@foods.each do |food|
     puts "#{food.name} : #{food.calory}kcal"
     all_calory += food.calory
    end

    puts line
    puts "カロリー合計 : #{all_calory}kcal"
   end

    

   def initialize(name, calory)
    @name = name
    @calory = calory
   end
end


  while true do
    puts "[0]:カロリーを入力する"
    puts "[1]:カロリーの合計を見る"
    input = gets.to_i

    if input == 0
      Food.input           # カロリーの入力
    elsif input == 1
      Food.show_all_calory # カロリーの合計を表示
      exit
    end
  end
