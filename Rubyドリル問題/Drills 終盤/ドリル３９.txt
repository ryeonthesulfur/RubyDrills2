■ ハッシュへの動的な代入
  student = {}                   → 空のハッシュを作ってから
  student[:name] = gets.chomp    → キーに直接値を代入していく書き方
  ハッシュリテラルで一括定義する書き方と同じ結果になる

■ 配列にハッシュを追加する
  students << student            → 完成したハッシュを配列に追加
  students は複数のハッシュを持つ配列になる

■ 登録 → 一覧 → 詳細の2メソッド構造
  registration_student → ハッシュ作成 → 配列追加
  show_student_name → 一覧表示 → 番号入力 → 詳細表示（1メソッド内で完結）
  詳細表示を別メソッドに切り出さず1つにまとめるパターンもある

■ 複数キーのハッシュ管理
  student[:name] / [:age] / [:japanese] / [:math] / [:english]
                                 → 1人分のデータを1ハッシュにまとめて管理

■ PHP 対応メモ
  student = {}                   → PHP では $student = []（PHPに {} の空ハッシュはない）
  student[:name] = ...           → PHP では $student['name'] = ...
  students << student            → PHP では $students[] = $student

■ Java 対応メモ
  students（配列）               → Java では ArrayList<String> など
  gets.chomp                     → Java では sc.nextLine()
  gets.to_i                      → Java では sc.nextInt()

function registration_student($students) {
  $student = {};
  echo "生徒名を入力してください"
  $student['name'] = trim(fgets(STDIN));
  echo "生徒の年齢を入力してください"
  $student['age'] = trim(fgets(STDIN));

  echo "国語の得点は？\n";
  $student['japanese'] = intval(trim(fgets(STDIN)));
  echo "数学の得点は？\n";
  $student['math'] = intval(trim(fgets(STDIN)));
  echo "英語の得点は？\n";
  $student['english'] = intval(trim(fgets(STDIN)));

  $students[] = $student;
}


function show_student_name($students) {
  echo "見たい生徒の番号を入力してください\n";
  foreach($students as $student => $index) {
    echo "$index : {$student['name']}\n";
  }

  $input = intval(trim(STDIN));
  $student = $students[$input - 1];

  echo "名前: {$student['name']}\n";
  echo "年齢: {$student['age']}\n";
  echo "国語: {$student['japanese']}\n";
  echo "数学: {$student['math']}\n";
  echo "英語: {$student['english']}\n";
}
