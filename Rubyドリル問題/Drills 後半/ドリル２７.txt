■ 平均値の計算パターン
  total = 0
  配列.each { |book| total += book[:price] }
  average = total / 配列.length → 合計 ÷ 個数 で平均を出す
  .round                        → 四捨五入（小数を整数に丸める）

■ 配列.length の使い所
  books.length                  → ハッシュを含む配列の要素数（本の冊数）を返す

■ 別メソッドへの切り出し
  def average_price(books)      → 平均計算を専用メソッドに切り出してスッキリさせる
  show_books の中から呼ぶ       → average_price(books) で呼び出して使う

■ round メソッド
  数値.round                    → 四捨五入して整数を返す
  数値.round(n)                 → 小数点 n 桁まで四捨五入

■ PHP 対応メモ
  配列.length                   → PHP では count($array)
  .round                        → PHP では round($value)
  total / books.length          → PHP では $total / count($books)

    $books[] = $book;
}

function show_books($books) {

    echo "見たい番号を入力してください\n";

    foreach($books as $index => $book) {

}

function average_price($books) {
    $total = 0;
    foreach($books as $book) {
        $total += $book['price'];
        $average = round($total / count($books));
    }
    return $average;
}
