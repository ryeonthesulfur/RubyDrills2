■ rand メソッド（乱数）
  rand(3)                        → 0, 1, 2 のどれかをランダムに返す
  rand(n) は 0〜n-1 の整数を返す

■ メソッドの return で while を制御するパターン
  def janken ... return true/false end
  next_game = janken             → メソッドの戻り値で while を継続するか決める
  あいこ → true → while 継続 / 勝ち負け → false → while 終了

■ じゃんけん勝利条件の書き方
  (player == 0 && program == 1)  → グーでチョキに勝つ
  || (player == 1 && program == 2) → チョキでパーに勝つ
  || (player == 2 && program == 0) → パーでグーに勝つ
  3パターンを || でつなぐ

■ 配列のインデックスで手を文字列に変換
  jankens = ["グー", "チョキ", "パー"]
  jankens[player_hand]           → 数値入力を文字列に変換して表示

■ PHP 対応メモ
  rand(3) （0〜2）               → PHP では rand(0, 2)（min, max を両方指定）
  return true/false              → PHP でも同じ書き方で動く

■ Java 対応メモ
  rand(3)                        → Java では new Random(); rand.nextInt(3)
  def janken                     → Java では public static boolean janken()

ヒント
乱数の作り方
ヒントとして、乱数の作り方を教えます。乱数を使うとプログラムの出す手をランダムにすることができます。


  rand(欲しい乱数の数)
randメソッドを使うと手軽に乱数を生成できます。randメソッドに引数として数を渡すと「0」から「渡した数 - 1」の間でランダムに数を返します。

  rand(100) # => 0〜99の中から１つ数字をランダムに返す
これを使えばランダムにじゃんけんの手を出すことができます。



【PHP版】

function janken() {
  echo "[0]:グー\n[1]:チョキ\n[2]:パー\n";

  $player_hand = intval(trim(fgets(STDIN)));
  $program_hand = rand(0, 2);

  $jankens = ["グー", "チョキ", "パー"];

  echo "あなたの手：" . $jankens[$player_hand] . ", わたしの手：" . $jankens[$program_hand] . "\n";

  if ($player_hand == $program_hand) {
    echo "あいこで\n";
    return true;
  } elseif (($player_hand == 0 && $program_hand == 1) 
      || ($player_hand == 1 && $program_hand == 2) 
      || ($player_hand == 2 && $program_hand == 0)) {
    echo "あなたの勝ちです\n";
    return false;

  } else {
    echo "あなたの負けです\n";
    return false;
  }
}

$next_game = true;

echo "最初はグー、じゃんけん...\n";

while ($next_game) {
  $next_game = janken();
}



【Java版】

import java.util.Scanner;
import java.util.Random;

public class Janken {

  public static boolean janken() {
    System.out.println("[0]:グー\n[1]:チョキ\n[2]:パー");

    Scanner sc = new Scanner(System.in);
    int player_hand = sc.nextInt();

    Random rand = new Random();
    int program_hand = rand.nextInt(3);

    String jankens[] = {"グー", "チョキ", "パー"};

    System.out.println(
      "あなたの手：" 
      + jankens[player_hand] 
      + "、わたしの手：" + jankens[program_hand]
    );

    if (player_hand == program_hand) {
      System.out.println("あいこで");
      return true;
    } else if ((player_hand == 0 && program_hand == 1) 
            || (player_hand == 1 && program_hand == 2) 
            || (player_hand == 2 && program_hand == 0)) {
      System.out.println("あなたの勝ちです。");
      return false;
    } else {
      System.out.println("あなたの負けです。");
      return false;
    }
  }
  
  public static void main(String[] args) {
    boolean next_game = true;
    while (next_game) {
      next_game = janken();
    }
  }
}
