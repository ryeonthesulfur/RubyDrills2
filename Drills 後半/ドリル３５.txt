【PHP版】

function register_data(&$train_info) {
    echo "車種を入力してください。\n";
    $train_name = trim(fgets(STDIN));
    echo "走行可能距離を入力してください。\n";
    $length = intval(trim(fgets(STDIN)));
    echo "乗車人数を入力してください。\n";
    $passengers = intval(trim(fgets(STDIN)));

    $info = ['train_name' => $train_name, 'length' => $length, 'passengers' => $passengers];
    $train_info[] = $info;
}

function confirm_data_list($train_info) {
    foreach($train_info as $index => $info) {
        echo ($index + 1) . ": {$info['train_name']}\n";
    }
    
    echo "確認したい番号を入力してください。\n";

    $input = intval(trim(fgets(STDIN)));
    confirm_data($train_info[$input - 1]);
}


function confirm_data($train_info) {
    echo "車種: {$train_info['train_name']}\n";
    echo "走行可能距離: {$train_info['length']}\n";
    echo "乗車人数: {$train_info['passengers']}\n";

    echo "これから走る距離(km/l)を入力してください。\n";
    $input_distance = intval(trim(fgets(STDIN)));

    calculate_fuel_consumption($train_info, $input_distance);
}


function calculate_fuel_consumption($train_info, $distance) {
        $fuel_consumption = $distance / $train_info['length'];
        echo "その目的地までは、ガソリンを" . round($fuel_consumption, 1) . "L消費します。\n";
}
